```python
def _fallback_gate_predict(self, texts: List[str]) -> Dict[str, np.ndarray]:
    """Fallback gate for no_gate_random or no_gate_confidence."""
    n = len(texts)
    if self.no_gate_mode == "random":
        # 随机决策：50%接受 ID，50% 拒识 OOS
        preds = np.random.binomial(1, 0.5, n).astype(np.int64)
    else:  # confidence
        # 基于Router置信度：高置信度→接受，低→拒识
        router_out = self._router_predict(texts)
        threshold = 0.5  # 可调参数
        preds = (np.array(router_out["domain_probs"]) < threshold).astype(np.int64)
    
    return {
        "pred": preds,
        "score": np.random.rand(n).astype(np.float32),
        "distance": np.random.rand(n).astype(np.float32),
        "radius": np.ones(n, dtype=np.float32),
        "margin_ok": np.ones(n, dtype=bool),
        "nearest_cluster": np.full(n, -1, dtype=np.int64),
        "nearest_intent": np.array([f"unknown_{i}" for i in range(n)], dtype=object),
    }
```

```python
class MultiPrototypeGate:
    def __init__(
        self,
        ...,
        scoring_mode: str = "max",  # "max", "topk_mean", "margin", "energy"
        topk: int = 1,  # for topk_mean
    ):
        self.scoring_mode = scoring_mode
        self.topk = topk
    
    def score_batch(self, embeddings: np.ndarray):
        """Return (score, nearest_intent) for each embedding."""
        similarities = self._compute_similarities(embeddings)  # shape: (n, num_intents)
        
        if self.scoring_mode == "max":
            scores = np.max(similarities, axis=1)
            intents = np.argmax(similarities, axis=1)
        
        elif self.scoring_mode == "topk_mean":
            topk_indices = np.argsort(similarities, axis=1)[:, -self.topk:]
            scores = np.mean(np.take_along_axis(similarities, topk_indices, axis=1), axis=1)
            intents = topk_indices[:, -1]  # 最高的那个
        
        elif self.scoring_mode == "energy":
            # Energy-based: logsumexp(-distance)
            scores = -np.log(np.sum(np.exp(-similarities), axis=1))
        
        elif self.scoring_mode == "margin":
            # max_sim - second_max_sim
            top2 = np.partition(similarities, -2, axis=1)[:, -2:]
            scores = top2[:, 1] - top2[:, 0]
            intents = np.argmax(similarities, axis=1)
        
        return list(zip(scores, intents))
```

```python
#!/usr/bin/env python3
"""Tune Gate threshold on validation set.

Protocol:
- Load gate predictions on val set
- Sweep threshold from 0.0 to 1.0
- For each threshold, compute macro_f1, oos_f1, gate_id_recall
- Save best threshold
- Output: threshold_tuning_results.json
"""

import json
import numpy as np
from sklearn.metrics import f1_score

def tune_threshold_grid(
    gate_scores: np.ndarray,
    true_labels: np.ndarray,  # 0=ID, 1=OOS
    intent_labels: np.ndarray,  # for macro_f1 computation
    grid_points: int = 101,
):
    """Grid search for best threshold."""
    thresholds = np.linspace(0.0, 1.0, grid_points)
    results = []
    
    best_macro_f1 = -1
    best_threshold = 0.5
    
    for threshold in thresholds:
        gate_preds = (gate_scores >= threshold).astype(int)
        
        # Compute metrics
        oos_f1 = f1_score(true_labels == 1, gate_preds == 1, zero_division=0)
        gate_id_recall = np.mean(gate_preds[true_labels == 0] == 0)  # Recall for ID class
        
        # Approximate macro_f1 (need full pipeline to be accurate)
        macro_f1_proxy = (oos_f1 + gate_id_recall) / 2  # Simplified
        
        results.append({
            "threshold": float(threshold),
            "oos_f1": float(oos_f1),
            "gate_id_recall": float(gate_id_recall),
            "macro_f1_proxy": float(macro_f1_proxy),
        })
        
        if macro_f1_proxy > best_macro_f1:
            best_macro_f1 = macro_f1_proxy
            best_threshold = float(threshold)
    
    return {
        "best_threshold": best_threshold,
        "best_macro_f1_proxy": float(best_macro_f1),
        "grid_sweep": results,
    }
```
